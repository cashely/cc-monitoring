```typescript
  const child = spawn('claude', spawnArgs, {

  const TIMEOUT = 5 * 60 * 1000 // 5 minutes
  const timeoutId = setTimeout(() => {

    if (!res.destroyed) {
      sendSSE('error', JSON.stringify({ message: 'Execution timeout (5 minutes)' }))
      sendSSE('done', JSON.stringify({ code: 1 }))
      res.end()
    }
  }, TIMEOUT)

  child.stdout?.on('data', (chunk) => {
    sendSSE('data', JSON.stringify({ type: 'stdout', content: chunk.toString() }))
  })

  child.stderr?.on('data', (chunk) => {
    sendSSE('data', JSON.stringify({ type: 'stderr', content: chunk.toString() }))
  })

  child.on('error', (err) => {
    clearTimeout(timeoutId)
    if (!res.destroyed) {
      sendSSE('error', JSON.stringify({ message: err.message }))
      sendSSE('done', JSON.stringify({ code: 1 }))
      res.end()
    }
  })

  child.on('close', (code) => {
    clearTimeout(timeoutId)
    if (!res.destroyed) {
      sendSSE('done', JSON.stringify({ code: code ?? 0 }))
      res.end()
    }
  })

  // Handle client disconnect
  res.on('close', () => {
    clearTimeout(timeoutId)
    if (!child.killed) {
      child.kill('SIGTERM')
```

```typescript
import type { ChatConversation } from '../types/chat'

const API_BASE = '/ui/api'

export const LOCALSTORAGE_KEY = 'cc-chat-conversations'

export function loadConversations(): ChatConversation[] {
  try {
    const raw = localStorage.getItem(LOCALSTORAGE_KEY)
    if (!raw) return []
    return JSON.parse(raw) as ChatConversation[]
  } catch {
    return []
  }
}

export function saveConversations(conversations: ChatConversation[]): void {
  localStorage.setItem(LOCALSTORAGE_KEY, JSON.stringify(conversations))
}

/**
 * 发送对话消息，建立 SSE 连接
 * 返回 EventSource 供调用方监听事件
 */
export function sendChatMessage(prompt: string, sessionId?: string): EventSource {
  // POST 请求无法直接用 EventSource（它只支持 GET）
  // 先发 POST 获取 SSE URL 或使用 fetch + ReadableStream
  // 这里用 fetch + 手动解析 SSE
  // 但 EventSource 更简单——我们需要改造为 GET 或用 fetch 方式

  // 实际方案：用 fetch 发 POST，手动解析 SSE stream
  // 返回一个自定义对象模拟 EventSource 行为
  const body = JSON.stringify({
    tool: 'claude',
    prompt,
    stream: true,
    sessionId
  })

  // 创建 AbortController 供取消
  const abortController = new AbortController()

  // 自定义 SSE 连接对象
  const sseConnection = {
    abortController,
    onSession: ((_sessionId: string) => {}) as (sid: string) => void,
    onData: ((_type: string, _content: string) => {}) as (type: string, content: string) => void,
    onError: ((_message: string) => {}) as (msg: string) => void,
    onDone: ((_code: number) => {}) as (code: number) => void,
    aborted: false
  }

  fetch(`${API_BASE}/exec`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body,
    signal: abortController.signal
  }).then(async (response) => {
    if (!response.ok) {
      const err = await response.json()
      sseConnection.onError(err.error || `HTTP ${response.status}`)
      return
    }

    const reader = response.body?.getReader()
    if (!reader) {
      sseConnection.onError('No response body')
      return
    }

    const decoder = new TextDecoder()
    let buffer = ''

    while (true) {
      const { done, value } = await reader.read()
      if (done || sseConnection.aborted) break

      buffer += decoder.decode(value, { stream: true })

      // 解析 SSE events
      const lines = buffer.split('\n')
      buffer = ''

      let currentEvent = ''
      let currentData = ''

      for (const line of lines) {
        if (line.startsWith('event: ')) {
          currentEvent = line.slice(7)
        } else if (line.startsWith('data: ')) {
          currentData = line.slice(6)
        } else if (line === '' && currentEvent && currentData) {
          // 完整 event，触发回调
          if (currentEvent === 'session') {
            sseConnection.onSession(currentData)
          } else if (currentEvent === 'data') {
            try {
              const parsed = JSON.parse(currentData)
              sseConnection.onData(parsed.type || 'stdout', parsed.content || '')
            } catch {
              // 非 JSON，当作 stdout
              sseConnection.onData('stdout', currentData)
            }
          } else if (currentEvent === 'error') {
            try {
              const parsed = JSON.parse(currentData)
              sseConnection.onError(parsed.message || currentData)
            } catch {
              sseConnection.onError(currentData)
            }
          } else if (currentEvent === 'done') {
            try {
              const parsed = JSON.parse(currentData)
              sseConnection.onDone(parsed.code ?? 0)
            } catch {
              sseConnection.onDone(0)
            }
          }
          currentEvent = ''
          currentData = ''
        } else if (line !== '' && !line.startsWith('event:') && !line.startsWith('data:')) {
          // 不完整行，保留到下次
          buffer = line
        }
      }
    }
  }).catch((err) => {
    if (!sseConnection.aborted) {
      sseConnection.onError(err.message || 'Connection failed')
    }
  })

  return sseConnection as unknown as EventSource  // 类型欺骗，实际返回 SSEConnection
}

// 更准确的返回类型
export interface SSEConnection {
  abortController: AbortController
  onSession: (sessionId: string) => void
  onData: (type: string, content: string) => void
  onError: (message: string) => void
  onDone: (code: number) => void
  abort: () => void
}

export function createChatConnection(prompt: string, sessionId?: string): SSEConnection {
  const abortController = new AbortController()

  const connection: SSEConnection = {
    abortController,
    onSession: () => {},
    onData: () => {},
    onError: () => {},
    onDone: () => {},
    abort: () => {
      abortController.abort()
    }
  }

  fetch(`${API_BASE}/exec`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      tool: 'claude',
      prompt,
      stream: true,
      sessionId
    }),
    signal: abortController.signal
  }).then(async (response) => {
    if (!response.ok) {
      const err = await response.json()
      connection.onError(err.error || `HTTP ${response.status}`)
      return
    }

    const reader = response.body?.getReader()
    if (!reader) {
      connection.onError('No response body')
      return
    }

    const decoder = new TextDecoder()
    let buffer = ''

    while (true) {
      const { done, value } = await reader.read()
      if (done) break

      buffer += decoder.decode(value, { stream: true })

      // 解析 SSE events
      const events = buffer.split('\n\n')
      buffer = events.pop() || ''  // 最后一段可能不完整

      for (const eventStr of events) {
        if (!eventStr.trim()) continue

        let eventType = ''
        let eventData = ''

        for (const line of eventStr.split('\n')) {
          if (line.startsWith('event: ')) {
            eventType = line.slice(7)
          } else if (line.startsWith('data: ')) {
            eventData = line.slice(6)
          }
        }

        if (!eventType || !eventData) continue

        if (eventType === 'session') {
          connection.onSession(eventData)
        } else if (eventType === 'data') {
          try {
            const parsed = JSON.parse(eventData)
            connection.onData(parsed.type || 'stdout', parsed.content || '')
          } catch {
            connection.onData('stdout', eventData)
          }
        } else if (eventType === 'error') {
          try {
            const parsed = JSON.parse(eventData)
            connection.onError(parsed.message || eventData)
          } catch {
            connection.onError(eventData)
          }
        } else if (eventType === 'done') {
          try {
            const parsed = JSON.parse(eventData)
            connection.onDone(parsed.code ?? 0)
          } catch {
            connection.onDone(0)
          }
        }
      }
    }
  }).catch((err) => {
    if (err.name !== 'AbortError') {
      connection.onError(err.message || 'Connection failed')
    }
  })

  return connection
}
```

```typescript
import { useState, useCallback, useRef } from 'react'
import { createChatConnection, loadConversations, saveConversations } from '../api/chat'
import type { ChatMessage, ChatConversation } from '../types/chat'

export function useChat() {
  const [conversations, setConversations] = useState<ChatConversation[]>(loadConversations())
  const [currentIndex, setCurrentIndex] = useState<number>(
    conversations.length > 0 ? conversations.length - 1 : -1
  )
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const connectionRef = useRef<ReturnType<typeof createChatConnection> | null>(null)

  const currentConversation = currentIndex >= 0 ? conversations[currentIndex] : null
  const messages = currentConversation?.messages ?? []
  const sessionId = currentConversation?.sessionId ?? undefined

  // 保存到 localStorage
  const persist = useCallback((updated: ChatConversation[]) => {
    saveConversations(updated)
    setConversations(updated)
  }, [])

  // 发送消息
  const sendMessage = useCallback((prompt: string) => {
    // 如果正在加载，不允许发送
    if (loading) return

    setLoading(true)
    setError(null)

    // 添加 user message
    const userMessage: ChatMessage = {
      id: crypto.randomUUID(),
      role: 'user',
      content: prompt,
      timestamp: Date.now()
    }

    let updatedConversations: ChatConversation[]
    let convIndex: number

    if (!sessionId) {
      // 新对话
      const newConv: ChatConversation = {
        sessionId: '',  // 临时为空，等 SSE session event 回填
        messages: [userMessage]
      }
      updatedConversations = [...conversations, newConv]
      convIndex = updatedConversations.length - 1
    } else {
      // 继续现有对话
      updatedConversations = [...conversations]
      updatedConversations[currentIndex] = {
        ...updatedConversations[currentIndex],
        messages: [...updatedConversations[currentIndex].messages, userMessage]
      }
      convIndex = currentIndex
    }

    // 添加空的 assistant message placeholder
    const assistantMessage: ChatMessage = {
      id: crypto.randomUUID(),
      role: 'assistant',
      content: '',
      timestamp: Date.now()
    }
    updatedConversations[convIndex] = {
      ...updatedConversations[convIndex],
      messages: [...updatedConversations[convIndex].messages, assistantMessage]
    }

    setCurrentIndex(convIndex)

    // 累积 stdout
    let accumulatedStdout = ''

    const connection = createChatConnection(prompt, sessionId)
    connectionRef.current = connection

    connection.onSession = (sid: string) => {
      // 回填 sessionId
      const patched = [...updatedConversations]
      patched[convIndex] = { ...patched[convIndex], sessionId: sid }
      persist(patched)
    }

    connection.onData = (type: string, content: string) => {
      if (type === 'stdout') {
        accumulatedStdout += content

        // 实时更新 assistant message content
        const patched = [...updatedConversations]
        const msgs = [...patched[convIndex].messages]
        msgs[msgs.length - 1] = {
          ...msgs[msgs.length - 1],
          content: accumulatedStdout
        }
        patched[convIndex] = { ...patched[convIndex], messages: msgs }
        persist(patched)
      }
      // stderr 不拼入 assistant content，忽略（后续可在 UI 中展示）
    }

    connection.onError = (message: string) => {
      setError(message)
      setLoading(false)
      connectionRef.current = null

      // 如果 assistant content 为空，移除 placeholder
      const patched = [...updatedConversations]
      if (!accumulatedStdout) {
        const msgs = patched[convIndex].messages.slice(0, -1)
        patched[convIndex] = { ...patched[convIndex], messages: msgs }
        persist(patched)
      }
    }

    connection.onDone = (code: number) => {
      setLoading(false)
      connectionRef.current = null

      if (code !== 0 && !accumulatedStdout) {
        // 进程非零退出且无输出，移除 assistant placeholder
        const patched = [...updatedConversations]
        const msgs = patched[convIndex].messages.slice(0, -1)
        patched[convIndex] = { ...patched[convIndex], messages: msgs }
        persist(patched)
      }
    }
  }, [conversations, currentIndex, sessionId, loading, persist])

  // 开始新对话
  const newConversation = useCallback(() => {
    // 如果正在加载，先取消
    if (connectionRef.current) {
      connectionRef.current.abort()
      connectionRef.current = null
      setLoading(false)
    }
    setCurrentIndex(-1)
    setError(null)
  }, [])

  return {
    messages,
    sessionId,
    loading,
    error,
    sendMessage,
    newConversation,
    conversations,
    currentIndex
  }
}
```

```tsx
import { useRef, useEffect } from 'react'
import { Bubble, Sender } from '@ant-design/x'
import { Button, Tooltip } from 'antd'
import { PlusOutlined } from '@ant-design/icons'
import { useTranslation } from 'react-i18next'
import { useChat } from '../hooks/useChat'

export function ChatUI(): JSX.Element {
  const { t } = useTranslation('app')
  const { messages, loading, error, sendMessage, newConversation } = useChat()
  const listRef = useRef<HTMLDivElement>(null)

  // 自动滚动到底部
  useEffect(() => {
    if (listRef.current) {
      listRef.current.scrollTop = listRef.current.scrollHeight
    }
  }, [messages])

  const handleSubmit = (content: string) => {
    if (!content.trim()) return
    sendMessage(content.trim())
  }

  const items = messages.map((msg) => ({
    key: msg.id,
    content: msg.content,
    role: msg.role as 'user' | 'assistant',
    loading: msg.role === 'assistant' && loading && !msg.content,
  }))

  return (
    <div className="bg-bg-card rounded-lg h-full flex flex-col">
      {/* 对话记录 */}
      <div
        ref={listRef}
        className="flex-1 overflow-y-auto px-4 py-2"
      >
        {messages.length === 0 && (
          <div className="flex items-center justify-center h-full text-text-muted text-sm">
            {t('chat.placeholder')}
          </div>
        )}
        <Bubble.List
          items={items}
          style={{ height: '100%' }}
        />
      </div>

      {/* 错误提示 */}
      {error && (
        <div className="px-4 py-1 text-text-primary text-xs bg-red-900/30">
          {t('chat.errorPrefix')}{error}
        </div>
      )}

      {/* 输入框区域 */}
      <div className="px-4 py-2 border-t border-border flex items-center gap-2">
        <Tooltip title={t('chat.newConversation')}>
          <Button
            type="text"
            icon={<PlusOutlined />}
            onClick={newConversation}
            size="small"
          />
        </Tooltip>
        <Sender
          onSubmit={handleSubmit}
          placeholder={t('chat.placeholder')}
          loading={loading}
          style={{ flex: 1 }}
        />
      </div>
    </div>
  )
}
```
